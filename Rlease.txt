    end
    return obj:FindFirstChild(objName)

      task.wait()

function AIMovementInLobby()
  local clock = 0
  local doneMoving = false
  _G.ai_done = false
  local zombie = LocalPlayer.Character
  if not zombie or not zombie:FindFirstChild("Humanoid") or not zombie:FindFirstChild("HumanoidRootPart") then
    _G.ai_done = true
    return
  end
  local Zombiehumanoid = zombie.Humanoid
  task.spawn(function()
    local randomWayPoints = {
      { Vector3.new(29.444, 10.68, 1005.28) },
      { Vector3.new(120.15, 5.05, 1005.26) },
      { Vector3.new(145.12, 4.79, 1028.20) },
      { Vector3.new(50.00, 5.40, 1025.59), Vector3.new(12.74, 4.94, 1046.24) },
      { Vector3.new(28.11, 4.82, 1041.24) }
    }

    local selected = randomWayPoints[math.random(1, #randomWayPoints)]
    for i, v in pairs(selected) do
      if _G.ai_done then break end
      local path = PathfindingService:CreatePath()
      pcall(function()
        path:ComputeAsync(zombie.HumanoidRootPart.Position, v)
        local waypoints = path:GetWaypoints()
        for _, waypoint in pairs(waypoints) do
          if _G.ai_done then break end
          Zombiehumanoid:MoveTo(waypoint.Position)
          Zombiehumanoid.MoveToFinished:Wait()
        end
      end)
    end
    doneMoving = true
  end)
  local maxWait = _G.maxWaitTimeInLobby or 3
  while not(doneMoving or clock > maxWait) do
    clock = clock + 0.1
    task.wait(0.1)
  end
  _G.ai_done = true
end

end

local function getMag(v1, v2)
  return (v1 - v2).magnitude
end

local function charLookAt(chr, target)
  if chr and chr.PrimaryPart and target and target.Position then
    local chrPos = chr.PrimaryPart.Position
    local tPos = target.Position
    chr:SetPrimaryPartCFrame(CFrame.new(Vector3.new(chrPos.X, chrPos.Y, chrPos.Z), Vector3.new(tPos.X, chrPos.Y, tPos.Z)))
  end
end

local function addHitBox(obj, size)
  pcall(function()
    local part = Instance.new("Part")
    cs:AddTag(part, "RayIgnore")
    part.Size = Vector3.new(size, size, size)
    part.CFrame = obj.PrimaryPart and obj.PrimaryPart.CFrame or obj.CFrame
    part.Name = "enemyRadius"
    part.Anchored = false
    part.CanCollide = false
    part.Transparency = 1
    part.Parent = obj
    local weld = Instance.new("WeldConstraint")
    weld.Part1 = obj.PrimaryPart or obj
    weld.Part0 = part
    weld.Parent = obj
  end)
end

local function _initHitBox(instance)
  if instance:IsA("Model") then
    task.spawn(function()
      local hrp = instance:WaitForChild("HumanoidRootPart", 5)
      local hum = instance:WaitForChild("Humanoid", 5)
      if hrp and hum then
        cs:AddTag(instance, "Enemy")
        if _G.doInstakill and instakillTable[instance.Name] then
          hum.Health = 0
        end
        addHitBox(instance, 7)
      end
    end)
  end
end

local function initHitBox()
  warn("[Dungeon Quest Reborn] Initializing Enemy Hitboxes...")
  if workspace:FindFirstChild("enemies") then
    workspace.enemies.ChildAdded:Connect(_initHitBox)
    for _, inst in pairs(workspace.enemies:GetChildren()) do
      _initHitBox(inst)
    end
  end
  if workspace:FindFirstChild("dungeon") then
    for _, room in pairs(workspace.dungeon:GetChildren()) do
      local efold = room:FindFirstChild("enemyFolder")
      if efold then
        efold.ChildAdded:Connect(_initHitBox)
        for _, inst in pairs(efold:GetChildren()) do
          _initHitBox(inst)
        end
      end
    end
    workspace.dungeon.ChildAdded:Connect(function(room)
      local efold = room:WaitForChild("enemyFolder", 5)
      if efold then
        efold.ChildAdded:Connect(_initHitBox)
        for _, inst in pairs(efold:GetChildren()) do
          _initHitBox(inst)
        end
      end
    end)
  end
end

local function findCloseEnemy()
  local bestMob = nil
  local bestDist = math.huge
  local _, _, _, root = getPlayer()
  if not root then return nil end

  local prio = cs:GetTagged("Prio-Enemy")
  if #prio > 0 then return prio[1] end

  local enemies = cs:GetTagged("Enemy")
  for _, mob in pairs(enemies) do
    if mob and mob.Parent and mob:FindFirstChild("HumanoidRootPart") and mob:FindFirstChild("Humanoid") and mob.Humanoid.Health > 0 then
      local dst = getMag(root.Position, mob.HumanoidRootPart.Position)
      if dst < bestDist then
        bestMob = mob
        bestDist = dst
      end
    end
  end
  return bestMob
end

local function followPath(targetPos)
  local _, chara, hum, root = getPlayer()
  if not hum or not root then return end
  hum:MoveTo(targetPos)
end

local function decide()
  enemy = findCloseEnemy()
  local _, chara, hum, root = getPlayer()
  if not root then return "nothing" end

  if enemy and enemy:FindFirstChild("HumanoidRootPart") then
    local dist = getMag(root.Position, enemy.HumanoidRootPart.Position)
    if dist > 6 then
      return "chase"
    else
      return "strafe"
    end
  end
  return "wait"
end

local function mainLoop()
  task.spawn(function()
    while not gameOver do
      local _, chara, hum, root = getPlayer()
      if root and hum and hum.Health > 0 then
        hum.WalkSpeed = walkspeed
        local thought = decide()
        if thought == "chase" and enemy and enemy:FindFirstChild("HumanoidRootPart") then
          followPath(enemy.HumanoidRootPart.Position)
          charLookAt(chara, enemy.HumanoidRootPart)
        elseif thought == "strafe" and enemy and enemy:FindFirstChild("HumanoidRootPart") then
          charLookAt(chara, enemy.HumanoidRootPart)
        end
      end
      if _G.extremelyFast then
        RunService.RenderStepped:Wait()
      else
        task.wait(0.1)
      end
    end
  end)
end

mainLoop()
